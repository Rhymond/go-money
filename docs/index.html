Quick Start

Create & display

// From smallest unit (precise)
price := money.New(10099, money.USD)    // $100.99

// From float (human-friendly)
price  = money.NewFromFloat(100.99, "USD")

price.Display()    // "$100.99"
price.AsFloat64()  // 100.99
price.Amount()     // 10099 (raw cents)
fmt.Println(price) // "$100.99"  (implements Stringer)

Arithmetic & percentage

net := money.NewFromFloat(49.99, "GBP")
vat := net.Percentage(20)               // 20% VAT = £10.00

gross, _ := net.Add(vat)                 // £59.99

discount := gross.Percentage(5)          // 5% discount = £3.00
final, _  := gross.Subtract(discount)   // £56.99

Penny-perfect allocation

bill := money.New(10085, "USD")    // $100.85

// Even split
shares, _ := bill.Split(3)
// $33.62, $33.62, $33.61  (total = $100.85 ✓)

// Ratio allocation: investor A 60%, B 30%, C 10%
parts, _ := bill.Allocate(60, 30, 10)
// $60.51, $30.26, $10.08  (total = $100.85 ✓)

Float multiply & country mapping

// Multiply by float (VAT, FX factor)
price := money.NewFromFloat(100.00, "USD")
withVAT := price.MultiplyFloat(1.21)    // $121.00

// Clamp to a range
min := money.New(500, "USD")
max := money.New(5000, "USD")
fee, _ := price.Clamp(min, max)           // $5.00 (clamped to min)

// Create from country code
m, _ := money.NewForCountry(1000, "JP")  // ¥1000 JPY
m, _ = money.NewForCountry(500, "DE")    // €5.00 EUR

code, _ := money.CurrencyForCountry("GB") // "GBP"

Aggregates

items := []*money.Money{
    money.New(999, "USD"),
    money.New(1499, "USD"),
    money.New(299, "USD"),
}

total, _  := money.Sum(items...)     // $27.97
min, _    := money.Min(items...)     // $2.99
max, _    := money.Max(items...)     // $14.99
avg, _    := money.Average(items...) // $9.32

Exchange Rates & Conversion

First-class currency conversion with precise rounding.

// Create a rate: 1 USD = 0.92 EUR
rate, err := money.NewExchangeRate("USD", "EUR", 0.92)

usd := money.New(1000, "USD")   // $10.00
eur, _ := rate.Convert(usd)      // €9.20

// Reverse direction
inv := rate.Invert()             // EUR → USD
usd2, _ := inv.Convert(eur)     // ~$10.00

// Rate metadata
rate.From()  // "USD"
rate.To()    // "EUR"
rate.Rate()  // 0.92

Rounding Modes

Choose the right strategy for taxes, accounting, and statistical work.

RoundHalfUp

Ties round away from zero. Most common for retail.

$1.50 → $2.00
$1.49 → $1.00

RoundHalfDown

Ties round toward zero.

$1.50 → $1.00
$1.51 → $2.00

RoundHalfEven

Banker's rounding — ties to nearest even. Minimises cumulative error.

$0.50 → $0.00 (even)
$1.50 → $2.00 (even)

RoundUp

Always round away from zero regardless of remainder.

$1.01 → $2.00
$1.99 → $2.00

RoundDown

Truncate toward zero — discard remainder.

$1.99 → $1.00
-$1.99 → -$1.00
m := money.New(150, "USD")   // $1.50

m.RoundWithMode(money.RoundHalfUp)   // $2.00
m.RoundWithMode(money.RoundHalfDown) // $1.00
m.RoundWithMode(money.RoundHalfEven) // $2.00  (2 is even)
m.RoundWithMode(money.RoundUp)       // $2.00
m.RoundWithMode(money.RoundDown)     // $1.00

// Banker's rounding on a batch:
money.New(250, "USD").RoundWithMode(money.RoundHalfEven) // $2.00 (even)
money.New(350, "USD").RoundWithMode(money.RoundHalfEven) // $4.00 (even)

SQL & JSON

JSON

// Encoding
m := money.New(1234, "USD")
b, _ := json.Marshal(m)
// {"amount":1234,"currency":"USD"}

// Decoding
var r money.Money
json.Unmarshal(b, &r)
r.Display() // "$12.34"

// Embedding in a struct
type Order struct {
    ID    int         `json:"id"`
    Price money.Money `json:"price"`
}

SQL (database/sql)

// Stored as "1234 USD" in any TEXT/VARCHAR column

// Writing
db.Exec("INSERT INTO orders (price) VALUES (?)", money.New(1234, "USD"))

// Reading
var price money.Money
row.Scan(&price)
price.Display() // "$12.34"
